```java
public abstract class DeviceSimulator {
    protected final String deviceNo;
    protected final String deviceType;
    protected final ScheduledExecutorService scheduler;   // 全局共享池

    /** 本设备登记的周期任务：stop() 只取消自己的，绝不动共享池 */
    private final List<ScheduledFuture<?>> periodicTasks = new CopyOnWriteArrayList<>();

    public void start() {
        schedule(this::safeSendHeartbeat, 0, heartbeatSeconds);
        schedule(this::safeSendPosition, 1, positionSeconds);
        onStarted();                                       // 子类注册特色任务
    }

    /** 指令幂等保护：同一条指令只执行一次 */
    public void handleCommand(CommandMessage cmd) {
        if (!processedMsgIds.add(cmd.msgId())) {
            log.warn("[{}] 指令 {} 已执行过，忽略重复投递", deviceNo, cmd.msgId());
            return;
        }
        scheduleOnce(() -> sendTaskAck(cmd.taskId(), "STARTED",  0, "任务开始执行"), 1);
        scheduleOnce(() -> sendTaskAck(cmd.taskId(), "PROGRESS", 50, "执行中"),       4);
        scheduleOnce(() -> sendTaskAck(cmd.taskId(), "FINISHED", 100, "任务执行完成"), 8);
    }
}
```

### A.2 平台业务服务——任务状态机（节选）

```java
public void applyAck(TaskAckMessage ack) {
    TaskDoc task = taskRepository.findById(ack.taskId()).orElse(null);
    if (task == null) { log.warn("[任务回执] 任务不存在: {}", ack.taskId()); return; }

    // 已终结的任务不再接受迟到回执，避免状态回退
    if (FINISHED.equals(task.getStatus()) || CANCELLED.equals(task.getStatus())
            || FAILED.equals(task.getStatus())) return;

    switch (ack.stage()) {
        case "STARTED"  -> { task.setStatus(RUNNING);  task.setProgress(0);   }
        case "PROGRESS" -> task.setProgress(ack.progress());
        case "FINISHED" -> { task.setStatus(FINISHED); task.setProgress(100);
                             task.setFinishTime(ack.eventTime()); }
        case "FAILED"   -> { task.setStatus(FAILED);   task.setFinishTime(ack.eventTime()); }
        default -> { return; }
    }
    task.setRemark(ack.remark());
    taskRepository.save(task);
}
```

### A.3 设备端影像上传（节选）

> **职责说明**：影像的**写入由设备自己完成**（`device-simulator` 的 `StorageClient`），
> 平台的 `HdfsService` 只负责**读取**（前端预览、下载）。这一分工是为了消除
> "平台需访问设备文件系统"这一不成立的前提，详见正文 5.2.3 与 5.3 问题 5。

```java
/** 上传本地文件到 HDFS，返回存储引用；失败返回 null */
public String upload(String localPath, String deviceNo, String fileId) {
    String dir = BASE_DIR + "/" + deviceNo + "/" + LocalDate.now().format(DAY);
    String hdfsPath = dir + "/" + fileId + ".jpg";

    if (!mkdirs(dir)) return null;                        // ① op=MKDIRS

    byte[] data = Files.readAllBytes(Paths.get(localPath));

    // ② op=CREATE → 期望 307 重定向
    HttpResponse<String> createResp = httpClient.send(createReq, HttpResponse.BodyHandlers.ofString());
    if (createResp.statusCode() != 307) return null;

    // ③ 手动跟随 Location，将真实数据 PUT 至 DataNode
    //    注意：Java HttpClient 默认 followRedirects=NEVER，必须手动处理
    String location = createResp.headers().firstValue("Location").orElse(null);
    if (location == null) return null;
    HttpResponse<String> putResp = httpClient.send(putReq, HttpResponse.BodyHandlers.ofString());
    return putResp.statusCode() == 201 ? hdfsPath : null;  // 返回存储引用
}
```

### A.5 权限拦截器（节选）

```java
@Override
public boolean preHandle(HttpServletRequest request, HttpServletResponse response,
                         Object handler) throws Exception {
    if ("OPTIONS".equalsIgnoreCase(request.getMethod())) return true;

    String auth = request.getHeader("Authorization");
    if (auth == null || !auth.startsWith("Bearer "))
        return reject(response, 401, "未登录，请先登录");

    Map<String, Object> claims = JwtUtil.parse(auth.substring(7));
    if (claims == null) return reject(response, 401, "登录已过期，请重新登录");

    String role = (String) claims.get("role");
    if (!hasPermission(request, role))
        return reject(response, 403, "当前角色无权执行该操作");

    request.setAttribute("currentUser", claims.get("username"));
    request.setAttribute("currentRole", role);
    return true;
}
```
